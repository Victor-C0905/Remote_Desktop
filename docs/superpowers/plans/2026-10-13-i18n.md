# Quirel 多语言（i18n）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 Quirel 增加可设置的语言功能（首期 zh-CN / en-US），语言包可增量扩展（新增语言 = 一个 TS 文件 + 注册表一行），Rust 错误消息彻底错误码化。

**Architecture:** react-i18next 内联资源初始化；词典以 zh-CN 为 source of truth，经 `TranslationSchema` 类型强制其他语言包结构完整；语言状态存 `settingsStore`（复用主题设置的持久化与生效路径）；Rust 端 `AppError{code,params,detail}` 结构化错误 + 前端查词典渲染。

**Tech Stack:** i18next + react-i18next、zustand persist（`@tauri-apps/plugin-store`）、Rust serde、vitest。

**需求来源（唯一）：** `docs/superpowers/specs/2026-10-13-i18n-design.md`

### 词典规则

1. 每语言一个单文件（`src/i18n/locales/<locale>.ts`）；按功能域一级分组，域内 camelCase；**词典不使用数组**（列表一律用固定 key 对象）。
2. `zh-CN.ts` 是 source of truth，导出 `as const` 对象；`src/i18n/types.ts` 从它推导 `TranslationSchema`，其他语言包标注该类型——缺 key / 多 key / 结构不一致直接编译报错。
3. zh 与 en 词典在**同一个 commit** 中更新：任何 key 两边同时加、同时改。
4. 插值用 `{{var}}`；i18next 配置 `interpolation.escapeValue: false`（React 已转义）。
5. **模块级常量不得缓存 `t()` 结果**（会缓存旧语言）：渲染函数内取值，或常量只存 key、渲染处 `t(key)`。
6. 中文文案搬入词典时逐字搬移、不改写；英文按术语表翻译。**宁可两个 key，不做语法拼接**（如 `上传完成/下载完成` 用两个 key，不用 `{{direction}}完成`）。

### 迁移模式手册（Phase 2 各任务引用，模式编号 P1~P8）

```tsx
// P1: JSX 文本 / 属性（title、placeholder、aria-label）
// before
<button title="后退">‹</button>
// after
<button title={t('fileManager.nav.back')}>‹</button>

// P2: 组件 props（含模板字符串 → 插值）
// before
<ConfirmDialog title="删除文件" message={`确定删除 "${name}"？`} />
// after（词典: "deleteConfirm": "确定删除 \"{{name}}\"？"）
<ConfirmDialog title={t('fileManager.deleteTitle')} message={t('fileManager.deleteConfirm', { name })} />

// P3: 非 React 纯 TS 模块（直接 import i18n 实例；项目无路径别名，用相对路径）
// before
alert('请先连接到远程服务器');
// after
import i18n from '../i18n';
alert(i18n.t('fileManager.notConnected'));

// P4: 插值格式化
// before
`${m}分${s}秒`
// after（词典: "duration": "{{m}}分{{s}}秒"）
i18n.t('transfer.duration', { m, s })

// P5: 条件文案（三元 → 两个 key，不做语法拼接）
// before
{direction === 'upload' ? '上传' : '下载'}
// after
{t(direction === 'upload' ? 'transfer.direction.upload' : 'transfer.direction.download')}

// P6: 系统通知（new Notification）
// before
new Notification('传输完成', { body: `${name} 已成功${direction}` });
// after
new Notification(i18n.t('transfer.notify.finishedTitle'), { body: i18n.t(direction === 'upload' ? 'transfer.notify.uploadDone' : 'transfer.notify.downloadDone', { name }) });

// P7: 模块级常量表（存 key，渲染时取值）
// before
const COLUMNS = [{ key: 'name', label: '名称' }];
// after
const COLUMNS = [{ key: 'name', labelKey: 'fileManager.columns.name' }];
// 渲染处：{t(col.labelKey)}

// P8: React 组件入口（useTranslation hook）
import { useTranslation } from 'react-i18next';
// 组件体内：
const { t } = useTranslation();
```

import { describe, it, expect } from 'vitest';
import i18n, { RESOURCES, FALLBACK_LOCALE } from '../index';

/** 提取词典对象的全部叶子路径（仅允许 string 叶子，词典不用数组） */
function shape(obj: unknown, path = ''): string[] {
  if (typeof obj === 'string') return [path];
  return Object.entries(obj as Record<string, unknown>).flatMap(([k, v]) =>
    shape(v, path ? `${path}.${k}` : k)
  );
}

describe('i18n 基础设施', () => {
  it('默认初始化为 zh-CN（fallback 也是 zh-CN）', () => {
    expect(i18n.language).toBe('zh-CN');
    expect(FALLBACK_LOCALE).toBe('zh-CN');
  });

  it('t() 可取到 common 域文案', () => {
    expect(i18n.t('common.confirm')).toBe('确定');
  });

  it('每种语言包结构与 zh-CN 完全一致（类型约束之外的运行时兜底）', () => {
    const zhKeys = shape(RESOURCES['zh-CN'].resource).sort();

    }
  });

  it('fallbackLng 配置为 zh-CN', () => {
    expect(i18n.options.fallbackLng).toBe('zh-CN');

});

- [ ] **Step 5: 创建 i18n 入口**

`src/i18n/index.ts`：

```ts
import i18next from 'i18next';
import { initReactI18next } from 'react-i18next';
import zhCN from './locales/zh-CN';
import enUS from './locales/en-US';

/**
 * 语言注册表。
 * 新增语言包协议（全项目唯一改动点）：
 *   1. src/i18n/locales/ 下新建 <locale>.ts，导出标注 TranslationSchema 的完整词典；
 *   2. 本注册表加一行 { label: '语言名', resource: 词典 }。
 * Settings 语言下拉框从 RESOURCES 渲染，新语言自动出现。
 */
export const RESOURCES = {
  'zh-CN': { label: '简体中文', resource: zhCN },
  'en-US': { label: 'English', resource: enUS },
} as const;

export type Locale = keyof typeof RESOURCES;
export const FALLBACK_LOCALE: Locale = 'zh-CN';

// 模块导入即初始化（默认 zh-CN），持久化的 locale 由 settingsStore rehydrate 后应用
void i18next.use(initReactI18next).init({
  resources: Object.fromEntries(
    Object.entries(RESOURCES).map(([code, { resource }]) => [code, { translation: resource }])
  ),
  lng: FALLBACK_LOCALE,
  fallbackLng: FALLBACK_LOCALE,
  interpolation: { escapeValue: false }, // React 已转义
  returnEmptyString: false,
});

/** 应用语言到 DOM 与 i18next（settingsStore 的 setLocale / onRehydrateStorage 调用） */
export function applyLocale(locale: Locale): void {
  document.documentElement.lang = locale;
  if (i18next.language !== locale) void i18next.changeLanguage(locale);
}

export default i18next;
```

  const { code, detail } = parseConnectError(err);

use serde::Serialize;
use std::collections::HashMap;

/// 用户可见错误的错误码常量（snake_case）。
/// 约束：新增只能追加；code 语义一经发布不可变更（前端词典按 code 查表）。
pub mod codes {
    pub const CONNECTION_NOT_FOUND: &str = "connection_not_found";
    pub const PERMISSION_DENIED: &str = "permission_denied";
    pub const FRAME_LENGTH_INVALID: &str = "frame_length_invalid";
    pub const FRAME_WRITE_FAILED: &str = "frame_write_failed";
    pub const FILE_NOT_FOUND: &str = "file_not_found";
    pub const IO_FAILED: &str = "io_failed";
    pub const AUTH_FAILED: &str = "auth_failed";
    pub const OPERATION_FAILED: &str = "operation_failed"; // 兜底
}

/// 结构化错误：invoke reject 时序列化为 { code, params?, detail? }。
/// - code：前端词典键（用户可见文案由此生成）
/// - params：用户需要的值（文件名/长度/端口），供前端插值
/// - detail：原始技术上下文（英文），仅日志排查用，前端不作为正文展示
#[derive(Debug, Clone, Serialize)]
pub struct AppError {
    pub code: String,
    #[serde(skip_serializing_if = "HashMap::is_empty")]
    pub params: HashMap<String, String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub detail: Option<String>,
}

impl AppError {
    pub fn new(code: &str) -> Self {
        Self { code: code.to_string(), params: HashMap::new(), detail: None }
    }

    pub fn with_param(mut self, key: &str, value: impl Into<String>) -> Self {
        self.params.insert(key.to_string(), value.into());
        self
    }

    pub fn with_detail(mut self, detail: impl Into<String>) -> Self {
        self.detail = Some(detail.into());
        self
    }
}

impl std::fmt::Display for AppError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        write!(f, "[{}] {}", self.code, self.detail.as_deref().unwrap_or(""))
    }
}

impl std::error::Error for AppError {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn 序列化结构为_code_params() {
        let err = AppError::new(codes::FRAME_LENGTH_INVALID)
            .with_param("length", "42")
            .with_detail("bad frame");
        let json = serde_json::to_string(&err).unwrap();
        assert_eq!(
            json,
            r#"{"code":"frame_length_invalid","params":{"length":"42"},"detail":"bad frame"}"#
        );
    }

    #[test]
    fn 空参数与空detail不序列化() {
        let err = AppError::new(codes::IO_FAILED);
        let json = serde_json::to_string(&err).unwrap();
        assert_eq!(json, r#"{"code":"io_failed"}"#);
    }

    #[test]
    fn display_便于日志排查() {
        let err = AppError::new(codes::IO_FAILED).with_detail("disk error");
        assert_eq!(err.to_string(), "[io_failed] disk error");
    }
}

/** Rust AppError 序列化形态（string code，区别于认证阶段 ConnectError 的数值 code） */
export interface OpError {
  code: string;
  params?: Record<string, string>;
  detail?: string;
}

export function isAppError(err: unknown): err is OpError {
  return (
    typeof err === "object" && err !== null && "code" in err &&
    typeof (err as { code: unknown }).code === "string"
  );
}

/** 字符串错误码 → 词典 key（errors.op.<key>）。与 src-tauri/src/errors.rs codes 模块一一对应 */
const OP_ERROR_KEY_MAP: Record<string, string> = {
  connection_not_found: "connectionNotFound",
  permission_denied: "permissionDenied",
  frame_length_invalid: "frameLengthInvalid",
  frame_write_failed: "frameWriteFailed",
  file_not_found: "fileNotFound",
  io_failed: "ioFailed",
  auth_failed: "authFailed",
  operation_failed: "operationFailed",
};

/** 格式化结构化操作错误为当前语言文案（未知 code 回退 detail 或 code 本身） */
export function formatOpError(err: OpError): string {
  const key = OP_ERROR_KEY_MAP[err.code];
  if (key) return i18n.t(`errors.op.${key}`, err.params ?? {});
  return err.detail || err.code;
}

/** 是否为权限不足（结构化 code 优先，保留字符串匹配兜底非迁移来源） */
export function isPermissionDenied(err: unknown): boolean {
  if (isAppError(err)) return err.code === "permission_denied";
  const msg = err instanceof Error ? err.message : String(err);
  return msg.includes("权限不足") || msg.toLowerCase().includes("permission denied");
}

  expect(isPermissionDenied('Permission denied')).toBe(true);
  expect(isPermissionDenied('other')).toBe(false);
});

# 新增语言包指南

export const RESOURCES = {
